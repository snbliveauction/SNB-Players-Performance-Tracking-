rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // অ্যাডমিন = যে Gmail বা ইমেইল-পাসওয়ার্ড দিয়ে সাইন ইন করেছে এবং admins collection এ যার ইমেইলের নামে একটা document আছে।
    function isAdmin() {
      return request.auth != null
        && (request.auth.token.email_verified == true
            || request.auth.token.firebase.sign_in_provider == 'password')
        && exists(/databases/$(database)/documents/admins/$(request.auth.token.email.lower()));
    }

    // প্লেয়ারদের তালিকা: সবাই দেখতে পারবে, শুধু অ্যাডমিন লিখতে পারবে।
    match /players/{id} {
      allow read: if true;
      allow create, update: if isAdmin()
        && request.resource.data.name is string
        && request.resource.data.name.size() <= 40;
      allow delete: if isAdmin();
    }

    // জয়ের হিসাব (মাস অনুযায়ী document, যেমন 2026-10)।
    match /months/{id} {
      allow read: if true;
      allow write: if isAdmin() && id.matches('^[0-9]{4}-[0-9]{2}$');
    }

    // অ্যাডমিন তালিকা: নিজের ইমেইলের document শুধু নিজে পড়তে পারে। লেখা যায় শুধু Firebase Console থেকে।
    match /admins/{email} {
      allow read: if request.auth != null && request.auth.token.email.lower() == email;
      allow write: if false;
    }
  }
}
